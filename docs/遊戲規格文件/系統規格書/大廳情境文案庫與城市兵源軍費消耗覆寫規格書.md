## 3. 軍事出征成本顯示覆寫契約

在任務卡片與任務詳情彈窗中，軍事任務（`MILITARY`）強制將通用體力代幣覆寫為實體軍政資源：

```typescript
export interface ITaskCostDisplay {
    readonly isMilitary: boolean;
    readonly requiredTroops: number;
    readonly requiredFood: number;
    readonly displayLabel: string;
    readonly canAfford: boolean;
    readonly errorHint: string | null;
}

export function formatMilitaryCostDisplay(
    isMilitary: boolean,
    requiredTroops: number,
    requiredFood: number,
    cityTroops: number,
    cityFood: number
): ITaskCostDisplay {
    if (!isMilitary) {
        return {
            isMilitary: false,
            requiredTroops: 0,
            requiredFood: 0,
            displayLabel: `消耗政務精力：${requiredFood} 點`,
            canAfford: true,
            errorHint: null,
        };
    }

    const hasTroops = cityTroops >= requiredTroops;
    const hasFood = cityFood >= requiredFood;

    let errorHint: string | null = null;
    if (!hasTroops) errorHint = '城中無兵可派（兵源不足）';
    else if (!hasFood) errorHint = '糧庫見底難以出征（糧草不足）';

    return {
        isMilitary: true,
        requiredTroops,
        requiredFood,
        displayLabel: `消耗城市兵源：${requiredTroops} 兵 ｜ 消耗庫存軍糧：${requiredFood} 石`,
        canAfford: hasTroops && hasFood,
        errorHint,
    };
}
```
